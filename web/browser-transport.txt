connectTransport(timeout=20000){this.onTransportState?.('connecting');return new Promise((resolve,reject)=>{const protocol=location.protocol==='https:'?'wss':'ws';const ws=new WebSocket(`${protocol}://${location.host}/game-socket?sala=${encodeURIComponent(window.cs16Sala||'')}`);ws.binaryType='arraybuffer';this.ws=ws;this.peer=ws;let ready=false;const timer=setTimeout(()=>{if(!ready){ws.close();reject(new Error('Game connection timed out. Please retry.'));}},timeout);ws.onmessage=e=>{if(typeof e.data==='string'){let m;try{m=JSON.parse(e.data);}catch{return;}if(m.type==='ready'){this.session=m.session;ready=true;clearTimeout(timer);this.channel={get readyState(){return ws.readyState===1?'open':'closed'},send:b=>{if(ws.readyState===1&&ws.bufferedAmount<1048576)ws.send(b)}};this.onTransportState?.('connected');resolve();}}else this.receive(e.data);};ws.onerror=()=>{if(!ready){clearTimeout(timer);reject(new Error('Could not reach this game room.'));}};ws.onclose=()=>{clearTimeout(timer);if(ready)this.lost(ws);else reject(new Error('Room unavailable or full. Return to the lobby.'));};});}
